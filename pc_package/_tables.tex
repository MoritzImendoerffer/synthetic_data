% Complex tables (Rev 2.0): pc_package/ctable.py emits multirow, booktabs and threeparttable.
\usepackage{multirow}
\usepackage{booktabs}
\usepackage[flushleft]{threeparttable}
\usepackage{adjustbox}
\usepackage{array}
\usepackage{pdflscape}   % landscape pages for the wide overview tables (CTable(landscape=True))
\usepackage{makecell}    % line breaks in header cells of non-wrapping columns
\usepackage{float}
\floatplacement{table}{H}   % complex tables stay where the text prints them (PCR-005 review)
% An uncaptioned longtable (a pipe table with no caption: related documents, procedures, registers)
% still steps the table counter, so the next captioned table skipped a number (Table 1.1 missing,
% PCR-005 review 2026-10-08). Give the number back when the table issued no caption.
% The patch is applied at the end of \begin{document}: the caption package redefines \LT@c@ption
% when the document begins, and a patch applied in the preamble was lost, so every longtable gave
% its number back and the table after a captioned pipe table repeated it (PCMP-001 printed two
% Table 1.1, PCR-005 final pass 2026-10-09). The number is given back only when both patches took.
\usepackage{etoolbox}
\makeatletter
\newif\ifLT@captioned
\newif\ifLT@patched
% The flag is set in \caption@LT@config, which the caption package runs inside a \noalign group
% for every longtable caption; a command placed before the \noalign of \LT@c@ption itself breaks
% the alignment ("Misplaced \noalign").
\AddToHook{begindocument/end}{%
  \pretocmd{\LT@array}{\global\LT@captionedfalse}{%
    \pretocmd{\caption@LT@config}{\global\LT@captionedtrue}{\global\LT@patchedtrue}{}}{}}
\AtEndEnvironment{longtable}{\ifLT@patched\ifLT@captioned\else\addtocounter{table}{-1}\fi\fi}
\makeatother
