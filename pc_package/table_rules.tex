% Variant rendering passes only (render_variant.py, make corpus-variant RULES=rows|grid). Loaded with
% --metadata-file table_rules.yml, before the document's own _tables.tex; the default pass never
% loads it.
%
% _tables.tex gives back the table number of an uncaptioned longtable (a pipe table with no
% caption) in an \AtEndEnvironment hook. The hook runs inside the alignment after the last row, so
% its \addtocounter opens an empty last row: a blank line above the bottom rule in the default pass,
% and in the grid a last row with broken vertical rules. table_rules.lua ends every pipe table with
% \PCLTgiveback, which gives the number back inside a \noalign and marks the table as captioned, so
% that the hook has nothing left to do. Without _tables.tex (no hook) it does nothing.
\makeatletter
\newcommand\PCLTgiveback{\noalign{\ifcsname ifLT@patched\endcsname\PC@LTgiveback\fi}}
\newcommand\PC@LTgiveback{\ifLT@patched\ifLT@captioned\else\addtocounter{table}{-1}\fi
  \global\LT@captionedtrue\fi}
\makeatother
